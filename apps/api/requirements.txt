# 3.3.0 was algorithm-confusable with non-HS keys (CVE-2024-33663) and
# DoS-able through jwe.decrypt (CVE-2024-33664). Neither was reachable
# here -- decode_access_token passes an explicit `algorithms=` allow-list
# and nothing calls jwe -- but "not reachable today" is not a pin.
# python-jose still drags in `ecdsa`, which has an unfixed Minerva timing
# flaw (CVE-2024-23342) its maintainers decline to fix; that is the open
# argument for moving to PyJWT, tracked in docs/progress/4.3.
python-jose[cryptography]==3.5.0

# Decision 0034: passlib is gone. It was unmaintained since 2020, and its
# bcrypt backend is the *only* reason this file used to pin
# `bcrypt==4.0.1` -- a security-critical library held backwards so a dead
# wrapper's start-up self-test would keep passing. argon2-cffi is now the
# hashing default; bcrypt stays as a verify-only path for credentials
# hashed before the change.
#   NOTE: these two lines and app/core/security.py must land together.
#   Installing this file against the old security.py fails on `import passlib`.
argon2-cffi==25.1.0
bcrypt==5.0.0

pyotp==2.9.0

# 43.0.1 shipped a statically linked OpenSSL with CVE-2024-12797, plus
# X.509 name-constraint bypasses (CVE-2026-69248/69249) and an ECDSA
# small-subgroup gap (CVE-2026-26007). This app only uses Fernet and
# HMAC, so none were reachable -- but Fernet is what every PHI column
# depends on, and this is not the dependency to be a version behind on.
cryptography==50.0.1

httpx==0.27.2

# Seven advisories against 0.0.9, all DoS/parser-differential in
# multipart and urlencoded form parsing. No route in this app declares a
# Form/File/UploadFile parameter, so Starlette never invokes the parser --
# unreachable today, and one `Form(...)` away from reachable.
python-multipart==0.0.32

# Phase 5.2. Error tracking. The API boots fine without this package and
# without a DSN -- that is the state in dev and in the test suite, and
# app/core/observability.py fails closed rather than degrading quietly: an
# SDK that rejects any of the PHI-safety kwargs gets no DSN at all.
# Pinned to the current release rather than the one the phase was written
# against, because include_local_variables has already been renamed once
# (from with_locals) and the init is asserted against the real kwarg names.
sentry-sdk==2.68.1
